clock = pygame.time.Clock()
last_fall_time = pygame.time.get_ticks()

fall_time = 0
fall_speed = 100

# Main game loop
running = True
while running:
    clock.tick(settings.GAME_SPEED)

    if game.block is None:
        game.new_block()

    current_time = pygame.time.get_ticks()
    if current_time - last_fall_time > fall_speed:
        last_fall_time = current_time
        game.go_down()

    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.KEYDOWN:

                game.go_down()
            if event.key == pygame.K_UP:

            if event.key == pygame.K_SPACE:
                game.go_space()

    screen.fill(settings.BLACK)

    for i in range(game.height):
        for j in range(game.width):

# tetris.py
from block import Block, shapes
import random


class Tetris:
    def __init__(self, height, width):
        self.height = height
        self.width = width
        self.board = [[0] * width for _ in range(height)]
        self.score = 0
        self.state = "start"
        self.block = None
        self.next_block = Block(random.choice(shapes))

    def new_block(self):
        self.block = self.next_block
        self.next_block = Block(random.choice(shapes))
        self.block.x = int(self.width / 2) - int(len(self.block.shape[0]) / 2)
        self.block.y = 0
        if self.intersect():
            self.state = "gameover"

    def intersect(self):
        for i in range(len(self.block.shape)):
            for j in range(len(self.block.shape[i])):
                if self.block.shape[i][j] != 0:
                    if i + self.block.y > self.height - 1 or \
                            j + self.block.x > self.width - 1 or \
                            j + self.block.x < 0 or \
                            self.board[i + self.block.y][j + self.block.x] != 0:
                        return True
        return False

    def freeze(self):
        for i in range(len(self.block.shape)):
            for j in range(len(self.block.shape[i])):
                if self.block.shape[i][j] != 0:
                    self.board[i + self.block.y][j + self.block.x] = self.block.shape[i][j]
        self.break_lines()
        self.new_block()

    def move(self, dx, dy):
        if self.block is None:
            return
        old_x = self.block.x
        old_y = self.block.y
        self.block.x += dx
        self.block.y += dy
        if self.intersect():
            self.block.x = old_x
            self.block.y = old_y

    def rotate(self):
        if self.block is None:
            return
        self.block.rotate()

    def break_lines(self):
        lines_to_remove = []
        for i, row in enumerate(self.board):
            if 0 not in row:
                lines_to_remove.append(i)
        for i in lines_to_remove:
            del self.board[i]
            self.board.insert(0, [0 for _ in range(self.width)])
        self.score += len(lines_to_remove)

    def go_space(self):
        while not self.intersect():
            self.block.y += 1
        self.block.y -= 1
        self.freeze()

    def go_down(self):
        if self.block is None:
            return
        self.block.y += 1
        if self.intersect():
            self.block.y -= 1
            self.freeze()
